\section{Numerické riešenie}
Keďže náš manipulátor sa skladá s 6 článkov, tak v nasledovných rovniciach budeme uvažovať 
výpočet pre týchto 6 článkov. Všeobecne však môžeme uvažovať ľubovoľný počet článkov,
a teda aj ľubovoľný počet premenných. Pseudo programy na výpočet týchto rovníc budú
preto všeobecne vyjadrené.

Pre výpočet budeme potrebovať základné vektory a matice. Budeme uvažovať polohový vektor:
\begin{equation}
\mathbf{p} = \begin{pmatrix} p^x \\ p^y \\ p^z \end{pmatrix}
\end{equation}
a vektor kĺbových premenných:
\begin{equation}
\boldsymbol{\theta} = \begin{pmatrix} \theta_1 \\ \theta_2 \\ \theta_3 \\ \theta_4 \\ \theta_5 \\ \theta_6 \end{pmatrix}
\end{equation}

rotačné matice a translačný vektor na výpočet polohy koncového efektora:
\begin{equation}
R_x(\theta) = \begin{pmatrix} 1 & 0 & 0 \\ 0 & \cos(\theta) & -\sin(\theta) \\ 0 & \sin(\theta) & \cos(\theta) \end{pmatrix}
\end{equation}

\begin{equation}
R_y(\theta) = \begin{pmatrix} \cos(\theta) & 0 & \sin(\theta) \\ 0 & 1 & 0 \\ -\sin(\theta) & 0 & \cos(\theta) \end{pmatrix}
\end{equation}

\begin{equation}
R_z(\theta) = \begin{pmatrix} \cos(\theta) & -\sin(\theta) & 0 \\ \sin(\theta) & \cos(\theta) & 0 \\ 0 & 0 & 1 \end{pmatrix}
\end{equation}

\begin{equation}
T = \begin{pmatrix} 0 \\ 0 \\ L \end{pmatrix}
\end{equation}

\subsection{Priama kinematika}
Priama kinematika nám umožňuje vypočítať polohu koncového efektora na základe zadaných kĺbových premenných.
Medzi jednotlivými súradnicovými systémami platia transformácie:
\begin{align}
p_5 &= R_6(\theta_6)(T_6 + p_6) \\
p_4 &= R_5(\theta_5)(T_5 + p_5) \\
p_3 &= R_4(\theta_4)(T_4 + p_4) \\
p_2 &= R_3(\theta_3)(T_3 + p_3) \\
p_1 &= R_2(\theta_2)(T_2 + p_2) \\
p_0 &= R_1(\theta_1)(T_1 + p_1)
\end{align}

Kinematika koncového efektora manipulátora je potom daná ako (transformácia z $p_6$ do $p_0$):
\begin{equation}
p_0(\theta) = R_1(\theta_1)\Bigg(T_1 + R_2(\theta_2)\Big(T_2 + R_3(\theta_3)\big(T_3 + R_4(\theta_4)\left(T_4 + R_5(\theta_5)\left(T_5 + R_6(\theta_6)(T_6)\right)\right)\big)\Big)\Bigg)
\end{equation}

Nižšie je uvedený pseudo program na výpočet priamej kinematikovej transformácie, ktorý zohľadňuje všetky články manipulátora a ich vzájomné transformácie.
Čiže pseudo program vracia polohový vektor a rotačnú maticu koncového efektora vzhľadom na základňu manipulátora.

\begin{lstlisting}[caption={Výpočet priamej kinematikovej transformácie}, language=C]
function forwardKinematics(theta[], R[])
    vekt = 0;
    matica = I_3x3;
    n = size(theta[]);
    
    for (i = n; i > 0; i--) begin
        vekt = R[i](theta[i]) * (T[i] + vekt);
        matica = R[i](theta[i]) * matica;
    end
    
    return vekt, matica;
end function
\end{lstlisting}

\subsection{Inverzná kinematika}
\subsubsection{Jakobián polohy}
Na výpočet inverznej kinematickej úlohy je potrebné poznať Jakobián polohy, ktorý nám umožní vypočítať zmenu polohy koncového efektora vzhľadom na zmenu kĺbových premenných.
Jakobiánu polohového vektora $p_0(\theta)$ možno definovať ako maticu parciálnych derivácií $\nabla p_0(\theta) \in \mathbb{R}^{3 \times 6}$:
\begin{equation}
\nabla p_0(\theta) = \begin{pmatrix} 
\frac{\partial p_0(\theta)}{\partial \theta_1} & \frac{\partial p_0(\theta)}{\partial \theta_2} & \frac{\partial p_0(\theta)}{\partial \theta_3} & \frac{\partial p_0(\theta)}{\partial \theta_4} & \frac{\partial p_0(\theta)}{\partial \theta_5} & \frac{\partial p_0(\theta)}{\partial \theta_6} 
\end{pmatrix}
\end{equation}

Nájdime teraz parciálne derivácie podľa kĺbových premenných:

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_1} = \frac{\partial(R_1(\theta_1))}{\partial \theta_1}\Bigg(T_1 + R_2(\theta_2)\Big(T_2 + R_3(\theta_3)\big(T_3 + R_4(\theta_4)\left(T_4 + R_5(\theta_5)\left(T_5 + R_6(\theta_6)(T_6)\right)\right)\big)\Big)\Bigg)
\end{equation}

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_2} = R_1(\theta_1)\Bigg(\frac{\partial(R_2(\theta_2))}{\partial \theta_2}\Big(T_2 + R_3(\theta_3)\big(T_3 + R_4(\theta_4)\left(T_4 + R_5(\theta_5)\left(T_5 + R_6(\theta_6)(T_6)\right)\right)\big)\Big)\Bigg)
\end{equation}

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_3} = R_1(\theta_1)R_2(\theta_2)\Bigg(\frac{\partial(R_3(\theta_3))}{\partial \theta_3}\Big(T_3 + R_4(\theta_4)\left(T_4 + R_5(\theta_5)\left(T_5 + R_6(\theta_6)(T_6)\right)\right)\Big)\Bigg)
\end{equation}

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_4} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)\Bigg(\frac{\partial(R_4(\theta_4))}{\partial \theta_4}\Big(T_4 + R_5(\theta_5)\left(T_5 + R_6(\theta_6)(T_6)\right)\Big)\Bigg)
\end{equation}

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_5} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)R_4(\theta_4)\Bigg(\frac{\partial(R_5(\theta_5))}{\partial \theta_5}\Big(T_5 + R_6(\theta_6)(T_6)\Big)\Bigg)
\end{equation}

\begin{equation}
\frac{\partial p_0(\theta)}{\partial \theta_6} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)R_4(\theta_4)R_5(\theta_5)\frac{\partial(R_6(\theta_6))}{\partial \theta_6}(T_6)
\end{equation}

\subsubsection{Pseudokód pre Jakobián}

\begin{lstlisting}[caption={Výpočet Jakobiánskej matice}, language=C]
function jacobian(theta[], R[], dR[])
    n = size(theta[]);
    jacob_mat = 3 x n;
    matica = I_3x3;
    pom_vekt = T[n];
    
    for(i = n-1; i > 0; i--)
    begin
        matica = R[i](theta[i]) * matica;
    end
    
    for(j = n; j > 0; j--)
    begin
        vekt = T[n];
        if(n > j)
        begin
            matica = matica * R[j](theta[j]) * transpose();
            pom_vekt = T[j] + R[j+1](theta[j+1]) * pom_vekt;
            vekt = pom_vekt;
        end
        
        vekt = matica + dR[j](theta[j]) * vekt;
        jacob_mat.col(j) = vekt;
    end
    
    return jacob_mat;
end function
\end{lstlisting}

\subsubsection{Jakobián orientácie}
Rotačné matice jednotlivých články robota tvoria celkovú rotačnú maticu pre súradnicový systém efektora, ktorú získame z funkcie priamej kinematiky. Táto matica nám umožní vypočítať orientáciu efektora vzhľadom na základňu manipulátora.
Celková rotačná matica $R_T(\theta)$ je daná súčinom rotačných matíc jednotlivých článkov:
\begin{equation}
R_T(\theta) = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)R_4(\theta_4)R_5(\theta_5)R_6(\theta_6)
\end{equation}

Nájdime teraz derivácie $R_T(\theta)$ vzhľadom na kĺbové premenné:

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_1} = \frac{\partial R_1(\theta_1)}{\partial \theta_1} R_2(\theta_2)R_3(\theta_3)R_4(\theta_4)R_5(\theta_5)R_6(\theta_6)
\end{equation}

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_2} = R_1(\theta_1) \frac{\partial R_2(\theta_2)}{\partial \theta_2} R_3(\theta_3)R_4(\theta_4)R_5(\theta_5)R_6(\theta_6)
\end{equation}

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_3} = R_1(\theta_1)R_2(\theta_2) \frac{\partial R_3(\theta_3)}{\partial \theta_3} R_4(\theta_4)R_5(\theta_5)R_6(\theta_6)
\end{equation}

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_4} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3) \frac{\partial R_4(\theta_4)}{\partial \theta_4} R_5(\theta_5)R_6(\theta_6)
\end{equation}

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_5} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)R_4(\theta_4) \frac{\partial R_5(\theta_5)}{\partial \theta_5} R_6(\theta_6)
\end{equation}

\begin{equation}
\frac{\partial R_T(\theta)}{\partial \theta_6} = R_1(\theta_1)R_2(\theta_2)R_3(\theta_3)R_4(\theta_4)R_5(\theta_5) \frac{\partial R_6(\theta_6)}{\partial \theta_6}
\end{equation}

\begin{itemize}
\item Definujeme reziduum (chybu) pre polohu efektora ako:
\begin{equation}
r_p(\theta) = p_0(\theta) - p_r
\end{equation}

\subsubsection{Pseudokód pre výpočet Jakobiánu orientácie}

\begin{lstlisting}[caption={Výpočet derivácií rotačných matíc}, language=C]
function derRotMat(theta[], dR[], R[])
    n = size(theta[]);
    mat_all(n);
    f_mat = I_3x3;
    b_mat = f_mat;
    
    for(i = n-1; i > 0; i--)
    begin
        f_mat = R[i](theta[i]) * matica;
    end
    
    for(j = n; j > 0; j--)
    begin
        if(n > j)
        begin
            b_mat = R[j+1](theta[j+1]) * b_mat;
            f_mat = f_mat * R[j](theta[j]).transpose();
        end
        mat_all(j) = f_mat * dR[j](theta[j]) * b_mat;
    end
    
    return mat_all;
end function
\end{lstlisting}

\item Definujeme reziduum (chybu) pre orientáciu efektora ako:
\begin{equation}
r_o(\theta) = \begin{pmatrix}
(0 \ 1 \ 0)(R_T(\theta)(R_{RPY}(\theta_R, \theta_P, \theta_Y))^T)\begin{pmatrix} 1 \\ 0 \\ 0 \end{pmatrix} \\
(0 \ 0 \ 1)(R_T(\theta)(R_{RPY}(\theta_R, \theta_P, \theta_Y))^T)\begin{pmatrix} 1 \\ 0 \\ 0 \end{pmatrix} \\
(0 \ 0 \ 1)(R_T(\theta)(R_{RPY}(\theta_R, \theta_P, \theta_Y))^T)\begin{pmatrix} 0 \\ 1 \\ 0 \end{pmatrix}
\end{pmatrix}
\end{equation}

\item Výsledkom riešenia by malo byť zabezpečiť že všetky rezidá sú nulové:
\begin{equation}
r_p(\theta) = 0
\end{equation}
\begin{equation}
r_o(\theta) = 0
\end{equation}

\item Krok Newtonovej metódy nech spĺňa sústavú lineárnych rovníc:
\begin{equation}
\scalebox{0.48}{

\end{equation}

\begin{equation}
\theta(k+1) = \theta(k) + \Delta\theta(k)
\end{equation}

\item Algoritmus ukončujeme ak je splnená podmienka (veľkosť rezidua):
\begin{equation}
(r_p(\theta))^T(r_p(\theta)) + (r_o(\theta))^T(r_o(\theta)) < \epsilon
\end{equation}
\end{itemize}

\subsubsection{Pseudokód pre inverznú kinematiku}

\begin{lstlisting}[caption={Iteratívny výpočet inverznej kinematiky}, language=C]
function solveIK(pr, Rr, theta, tol_pos, tol_ori, max_iter)
    for(i=0; i < max_iter; i++)
    begin
        [p0, R0] = forwardKinematics(theta, R);
        (3,3) rp = p0 - Pr;
        (3,3) E = R0 + Rr.transpose();
        ro(0,0) = E(1,0);                         // (010) * E(100)^T
        ro(1,0) = E(2,0);                         // (001) * E(100)^T
        ro(2,0) = E(2,1);                         // (001) * E(010)^T
        
        if(rp.norm < tol_pos && ro.norm < tol_ori)
        begin
            return 1;                              // úspešne max_iterov
        end
        
        J_p = jakobian(theta, R, dR);
        der_R = derRotMat(theta, dR, R);
        J.block<3,6>(0,0) = J_p.block<3,6>(0,0);

        for(j=0; j < DOF; ++j)
        begin
        (3,3) dE_j = der_R[j] * Rr.transpose();

        J(3,j) = dE_j(1,0);
        J(4,j) = dE_j(2,0);
        J(5,j) = dE_j(2,1);
        end
        
        for(k=0; k < DOF; ++k)
        begin
            J(k,k) += lambda;
        end

        minus_r.block<3,1>(0,0) = -rp;
        minus_r.block<3,1>(3,0) = -ro;
        delta_theta = vyries_sustavu_6x6(J, minus_r);
        theta = theta + delta_theta; 
    end
    return false;   
end function
\end{lstlisting}
