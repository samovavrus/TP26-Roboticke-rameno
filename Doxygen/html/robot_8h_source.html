249 std::pair<Matrix<3,1>, Matrix<3,3>>
250 forwardKinematics(const Matrix<DOF,1>& theta) const override
251 {
252 // Iterujeme od posledného kĺbu dovnútra smerom k základni
253 Matrix<3,1> vekt = Matrix<3,1>::Zero();
254 for (int i = (int)DOF - 1; i >= 0; --i)
255 {
256 vekt = (*R_func[i])(theta(i)) * (L[i] + vekt);
257 }
258
259 // Orientácia efektora: R1 * R2 * ... * Rn
260 Matrix<3,3> R_ee = Matrix<3,3>::Identity();
261 for (std::size_t i = 0; i < DOF; ++i)
262 R_ee = R_ee * (*R_func[i])(theta(i));
263
264 return { vekt, R_ee };
265 }

276 std::array<Matrix<3, 3>, DOF> getRotDerivatives(const Matrix<DOF, 1>& theta) const {
277 std::array<Matrix<3, 3>, DOF> mat_all; // Pole pre výsledné parciálne derivácie
278 int n = DOF;
279 Matrix<3, 3> f_mat = Matrix<3, 3>::Identity();
280 Matrix<3, 3> b_mat = Matrix<3, 3>::Identity();
281
282 // Prvý cyklus: Výpočet celkového dopredného súčinu matíc
283 // Týmto f_mat naakumuluje súčin R_0 * R_1 * ... * R_{n-2}
284 for (int i = n - 2; i >= 0; i--) {
285 f_mat = (*R_func[i])(theta(i)) * f_mat;
286 }
287
288 // Druhý cyklus: Výpočet derivácií smerom od konca k začiatku
289 for (int j = n - 1; j >= 0; j--) {
290 if (j < n - 1) {
291 b_mat = (*R_func[j + 1])(theta(j + 1)) * b_mat;
292
293 // "Odstraňovanie" aktuálnej matice zľava pomocou transpozície (R^-1 = R^T)
294 f_mat = f_mat * (*R_func[j])(theta(j)).transpose();
295 }
296
297 // Samotný výpočet derivácie pre j-ty kĺb
298 // Pre prístup k derivácii voláme metódu d() cez pointer z tvojho poľa R_func
299 mat_all[j] = f_mat * R_func[j]->d(theta(j)) * b_mat;
300 }
301
302 return mat_all;
303}

316 Matrix<3, DOF> getJacobian(const Matrix<DOF, 1>& theta) const {
317 int n = DOF; // Počet kĺbov
318 Matrix<3, DOF> jakob_mat = Matrix<3, DOF>::Zero(); // Výsledná Jakobián matica (3 riadky, n stĺpcov)
319
320 Matrix<3, 3> matica = Matrix<3, 3>::Identity(); // [1 0 0; 0 1 0; 0 0 1]
321 Matrix<3, 1> pom_vekt = L[n - 1]; // Inicializácia T[n] v pseudokóde
322
323 // Prvý cyklus: od i = n-1 do 1 (1-based) -> od i = n-2 po 0 (0-based)
324 // Kumuluje doprednú kinematiku (súčin rotácií od n-2 po 0)
325 for (int i = n - 2; i >= 0; i--) {
326 matica = (*R_func[i])(theta(i)) * matica;
327 }
328
329 // Druhý cyklus: od i = n do 1 (1-based) -> od i = n-1 po 0 (0-based)
330 for (int i = n - 1; i >= 0; i--) {
331
332 Matrix<3, 1> vekt = L[n - 1]; // T[n] z pseudokódu
333
334 if (i < n - 1) { // Podmienka (n > i) z pseudokódu v 0-based svete
335
336 // "Odstraňovanie" rotácie pre aktuálny kĺb
337 matica = matica * (*R_func[i])(theta(i)).transpose();
338
339 // Akumulácia translačných vektorov od konca (efektora) ku kĺbu 'i'
340 pom_vekt = L[i] + (*R_func[i + 1])(theta(i + 1)) * pom_vekt;
341
342 vekt = pom_vekt;
343 }
344
345 // Výpočet samotného stĺpca Jakobiánu
346 // matica: (R_1 * R_2 ... * R_{i-1})
347 // dR_i: parciálna derivácia i-teho kĺbu
348 // vekt: (T_i + R_{i+1} * (T_{i+1} + ...))
349 vekt = matica * R_func[i]->d(theta(i)) * vekt;
350
351 // Uloženie výsledného vektora do i-teho stĺpca matice Jakobiánu
352 jakob_mat.col(i) = vekt;
353 }
354
355 return jakob_mat;
356}

379 bool SolveIK(
380 const Matrix<3, 1>& pr,
381 const Matrix<3, 3>& Rr,
382 Matrix<DOF, 1>& theta,
383 float tol_pos,
384 float tol_ori,
385 float lambda,
386 int max_iter) override
387 {
388 for (int i = 0; i < max_iter; i++)
389 {
390 // 1. Dopredná kinematika (aktuálny stav)
391 auto fk_res = forwardKinematics(theta);
392 Matrix<3, 1> p0 = fk_res.first;
393 Matrix<3, 3> R0 = fk_res.second;
394
395 // 2. Výpočet reziduí (chyby)
396 Matrix<3, 1> rp = p0 - pr;
397
398 // Výpočet matice chyby orientácie E = R0 * Rr^T
399 Matrix<3, 3> E = R0 * Rr.transpose();
400
401 // Extrakcia chyby orientácie podľa tvojho pseudokódu
402 Matrix<3, 1> r0;
403 r0(0) = E(1, 0);
404 r0(1) = E(2, 0);
405 r0(2) = E(2, 1);
406
407 // 3. Kontrola ukončenia (ak sme v tolerancii)
408 if (rp.norm() < tol_pos && r0.norm() < tol_ori) {
409 return true;
410 }
411
412 // 4. Získanie Jakobiánov (tieto funkcie už v robot.h máš)
413 Matrix<3, DOF> J_p = getJacobian(theta);
414 std::array<Matrix<3, 3>, DOF> dEr_R = getRotDerivatives(theta);
415
416 // Zostavenie plného Jakobiánu 6xDOF
417 Matrix<6, DOF> J = Matrix<6, DOF>::Zero();
418
419 // Horná časť (poloha)
420 J.template block<3, DOF>(0, 0) = J_p;
421
422 // Dolná časť (orientácia - podľa dE_j = dR_j * Rr^T)
423 for (std::size_t j = 0; j < DOF; j++) {
424 Matrix<3, 3> dE_j = dEr_R[j] * Rr.transpose();
425
426 J(3, j) = dE_j(1, 0);
427 J(4, j) = dE_j(2, 0);
428 J(5, j) = dE_j(2, 1);
429 }
430
431 // 5. Regularizácia (Levenberg-Marquardt štýl)
432 for (std::size_t k = 0; k < DOF; k++) {
433 J(k, k) += lambda;
434 }
435
436 // 6. Zostavenie vektora záporných reziduí
437 Matrix<6, 1> minus_r;
438 minus_r.template block<3, 1>(0, 0) = -rp;
439 minus_r.template block<3, 1>(3, 0) = -r0;
440
441 // 7. Výpočet zmeny uhlov (Riešenie sústavy J * delta_theta = minus_r)
442 // Pre 6x6 maticu je ColPivHouseholderQR veľmi stabilná voľba
443 Matrix<DOF, 1> delta_theta = J.colPivHouseholderQr().solve(minus_r);
444
445 // 8. Update
446 theta += delta_theta;
447 }
448
449 return false; // Nepodarilo sa nájsť riešenie v rámci max_iter
450 }

453private:
454 std::array<Matrix<3, 1>, DOF> L;
455 std::array<const RotationMatrix*, DOF> R_func;
456};
