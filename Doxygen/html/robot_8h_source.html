260 std::pair<Matrix<3,1>, Matrix<3,3>>
261 forwardKinematics(const Matrix<DOF,1>& theta) const override
262 {
263 // Iterujeme od posledného kĺbu dovnútra smerom k základni
264 Matrix<3,1> vekt = Matrix<3,1>::Zero();
265 for (int i = (int)DOF - 1; i >= 0; --i)
266 {
267 vekt = (*R_func[i])(theta(i)) * (L[i] + vekt);
268 }
269
270 // Orientácia efektora: R1 * R2 * ... * Rn
271 Matrix<3,3> R_ee = Matrix<3,3>::Identity();
272 for (std::size_t i = 0; i < DOF; ++i)
273 R_ee = R_ee * (*R_func[i])(theta(i));
274
275 return { vekt, R_ee };
276 }

287 std::array<Matrix<3, 3>, DOF> getRotDerivatives(const Matrix<DOF, 1>& theta) const {
288 std::array<Matrix<3, 3>, DOF> mat_all; // Pole pre výsledné parciálne derivácie
289 int n = DOF;
290 Matrix<3, 3> f_mat = Matrix<3, 3>::Identity();
291 Matrix<3, 3> b_mat = Matrix<3, 3>::Identity();
292
293 // Prvý cyklus: Výpočet celkového dopredného súčinu matíc
294 // Týmto f_mat naakumuluje súčin R_0 * R_1 * ... * R_{n-2}
295 for (int i = n - 2; i >= 0; i--) {
296 f_mat = (*R_func[i])(theta(i)) * f_mat;
297 }
298
299 // Druhý cyklus: Výpočet derivácií smerom od konca k začiatku
300 for (int j = n - 1; j >= 0; j--) {
301 if (j < n - 1) {
302 b_mat = (*R_func[j + 1])(theta(j + 1)) * b_mat;
303
304 // "Odstraňovanie" aktuálnej matice zľava pomocou transpozície (R^-1 = R^T)
305 f_mat = f_mat * (*R_func[j])(theta(j)).transpose();
306 }
307
308 // Samotný výpočet derivácie pre j-ty kĺb
309 // Pre prístup k derivácii voláme metódu d() cez pointer z tvojho poľa R_func
310 mat_all[j] = f_mat * R_func[j]->d(theta(j)) * b_mat;
311 }
312
313 return mat_all;
314}

327 Matrix<3, DOF> getJacobian(const Matrix<DOF, 1>& theta) const {
328 int n = DOF; // Počet kĺbov
329 Matrix<3, DOF> jakob_mat = Matrix<3, DOF>::Zero(); // Výsledná Jakobián matica (3 riadky, n stĺpcov)
330
331 Matrix<3, 3> matica = Matrix<3, 3>::Identity(); // [1 0 0; 0 1 0; 0 0 1]
332 Matrix<3, 1> pom_vekt = L[n - 1]; // Inicializácia T[n] v pseudokóde
333
334 // Prvý cyklus: od i = n-1 do 1 (1-based) -> od i = n-2 po 0 (0-based)
335 // Kumuluje doprednú kinematiku (súčin rotácií od n-2 po 0)
336 for (int i = n - 2; i >= 0; i--) {
337 matica = (*R_func[i])(theta(i)) * matica;
338 }
339
340 // Druhý cyklus: od i = n do 1 (1-based) -> od i = n-1 po 0 (0-based)
341 for (int i = n - 1; i >= 0; i--) {
342
343 Matrix<3, 1> vekt = L[n - 1]; // T[n] z pseudokódu
344
345 if (i < n - 1) { // Podmienka (n > i) z pseudokódu v 0-based svete
346
347 // "Odstraňovanie" rotácie pre aktuálny kĺb
348 matica = matica * (*R_func[i])(theta(i)).transpose();
349
350 // Akumulácia translačných vektorov od konca (efektora) ku kĺbu 'i'
351 pom_vekt = L[i] + (*R_func[i + 1])(theta(i + 1)) * pom_vekt;
352
353 vekt = pom_vekt;
354 }
355
356 // Výpočet samotného stĺpca Jakobiánu
357 // matica: (R_1 * R_2 ... * R_{i-1})
358 // dR_i: parciálna derivácia i-teho kĺbu
359 // vekt: (T_i + R_{i+1} * (T_{i+1} + ...))
360 vekt = matica * R_func[i]->d(theta(i)) * vekt;
361
362 // Uloženie výsledného vektora do i-teho stĺpca matice Jakobiánu
363 jakob_mat.col(i) = vekt;
364 }
365
366 return jakob_mat;
367}

390 bool SolveIK(
391 const Matrix<3, 1>& pr,
392 const Matrix<3, 3>& Rr,
393 Matrix<DOF, 1>& theta,
394 float tol_pos,
395 float tol_ori,
396 float lambda,
397 int max_iter) override
398 {
399 for (int i = 0; i < max_iter; i++)
400 {
401 // 1. Dopredná kinematika (aktuálny stav)
402 auto fk_res = forwardKinematics(theta);
403 Matrix<3, 1> p0 = fk_res.first;
404 Matrix<3, 3> R0 = fk_res.second;
405
406 // 2. Výpočet reziduí (chyby)
407 Matrix<3, 1> rp = p0 - pr;
408
409 // Výpočet matice chyby orientácie E = R0 * Rr^T
410 Matrix<3, 3> E = R0 * Rr.transpose();
411
412 // Extrakcia chyby orientácie podľa tvojho pseudokódu
413 Matrix<3, 1> r0;
414 r0(0) = E(1, 0);
415 r0(1) = E(2, 0);
416 r0(2) = E(2, 1);
417
418 // 3. Kontrola ukončenia (ak sme v tolerancii)
419 if (rp.norm() < tol_pos && r0.norm() < tol_ori) {
420 return true;
421 }
422
423 // 4. Získanie Jakobiánov (tieto funkcie už v robot.h máš)
424 Matrix<3, DOF> J_p = getJacobian(theta);
425 std::array<Matrix<3, 3>, DOF> dEr_R = getRotDerivatives(theta);
426
427 // Zostavenie plného Jakobiánu 6xDOF
428 Matrix<6, DOF> J = Matrix<6, DOF>::Zero();
429
430 // Horná časť (poloha)
431 J.template block<3, DOF>(0, 0) = J_p;
432
433 // Dolná časť (orientácia - podľa dE_j = dR_j * Rr^T)
434 for (std::size_t j = 0; j < DOF; j++) {
435 Matrix<3, 3> dE_j = dEr_R[j] * Rr.transpose();
436
437 J(3, j) = dE_j(1, 0);
438 J(4, j) = dE_j(2, 0);
439 J(5, j) = dE_j(2, 1);
440 }
441
442 // 5. Regularizácia (Levenberg-Marquardt štýl)
443 for (std::size_t k = 0; k < DOF; k++) {
444 J(k, k) += lambda;
445 }
446
447 // 6. Zostavenie vektora záporných reziduí
448 Matrix<6, 1> minus_r;
449 minus_r.template block<3, 1>(0, 0) = -rp;
450 minus_r.template block<3, 1>(3, 0) = -r0;
451
452 // 7. Výpočet zmeny uhlov (Riešenie sústavy J * delta_theta = minus_r)
453 // Pre 6x6 maticu je ColPivHouseholderQR veľmi stabilná voľba
454 Matrix<DOF, 1> delta_theta = J.colPivHouseholderQr().solve(minus_r);
455
456 // 8. Update
457 theta += delta_theta;
458
459 // --- PRIDANÉ: Normalizácia uhlov do intervalu [-pi, pi] na záver iterácie ---
460 for (std::size_t j = 0; j < DOF; ++j) {
461 // std::remainder vráti zvyšok po delení, centrovaný okolo 0
462 theta(j) = std::remainder(theta(j), 2.0f * M_PI);
463 }
464 // --------------------------------------------------------------------------
465 }
466
467 return false; // Nepodarilo sa nájsť riešenie v rámci max_iter
468 }
